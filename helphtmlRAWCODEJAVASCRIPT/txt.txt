            function actualizarBloqueo() {
                const boton = document.getElementById("btnEnviar");

                const ultimoEnvio = localStorage.getItem("ultimoEnvio");

                if (!ultimoEnvio) return;

                const tiempoRestante =
                    100000 - (Date.now() - Number(ultimoEnvio));

                if (tiempoRestante <= 0) {
                    localStorage.removeItem("ultimoEnvio");
                    boton.disabled = false;
                    boton.textContent = "Send";
                    return;
                }

                boton.disabled = true;

                const intervalo = setInterval(() => {
                    const restante =
                        100000 - (Date.now() - Number(ultimoEnvio));

                    if (restante <= 0) {
                        clearInterval(intervalo);
                        boton.disabled = false;
                        boton.textContent = "Send";
                        localStorage.removeItem("ultimoEnvio");
                        return;
                    }

                    boton.textContent = `Wait ${Math.ceil(restante / 1000)}s...`;
                }, 1000);
            }

            async function enviar() {
                const texto = document.getElementById("mensaje").value.trim();
                const gmail = document.getElementById("gmailobg").value.trim();

                if (!texto) {
                    alert("Please write a message.");
                    return;
                }

                if (!gmail) {
                    alert("Please provide your email if you want a response.");
                    return;
                }

                const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

                if (!emailRegex.test(gmail)) {
                    alert("Please enter a valid email address.");
                    return;
                }

                const { error } = await supabase
                    .from("Test1")
                    .insert([
                        {
                            talk: texto,
                            recived_talk: true,
                            gmailobg: gmail
                        }
                    ]);

                if (error) {
                    console.error("Supabase error:", error);
                    alert(error.message);
                    return;
                }

                document.getElementById("mensaje").value = "";
                document.getElementById("gmailobg").value = "";

                localStorage.setItem("ultimoEnvio", Date.now());

                actualizarBloqueo();

                alert("Sent correctly. Thank you for using SoPrimico!");
            }

            document
                .getElementById("btnEnviar")
